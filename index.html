class FounderKernelEngine:
    """
    Real-time mathematical runway sentry and runaway API circuit breaker.
    Guarantees solo-founder operational solvency against recursive loops.
    """
    def __init__(self, initial_cash_reserve: float, fixed_daily_burn: float = 5.0):
        self.cash_reserve = float(initial_cash_reserve)
        self.fixed_daily_burn = float(fixed_daily_burn)
        self.cumulative_variable_spend = 0.0
        self.audit_log = []
        self.window_spend = []
        self.current_tier = "TIER_0_OPTIMAL"

    def calculate_runway(self) -> dict:
        now = time.time()
        recent_spend = sum(c for ts, c in self.window_spend if now - ts <= 86400)
        total_daily_burn = self.fixed_daily_burn + recent_spend
        runway_days = self.cash_reserve / total_daily_burn if self.cash_reserve > 0 else 0.0

        if runway_days >= 60.0:
            self.current_tier = "TIER_0_OPTIMAL"
        elif runway_days >= 30.0:
            self.current_tier = "TIER_1_THROTTLED"
        elif runway_days >= 14.0:
            self.current_tier = "TIER_2_DOWNGRADED"
        else:
            self.current_tier = "TIER_3_HARD_HALT"

        return {"runway_days": runway_days, "tier": self.current_tier}

    def evaluate_transaction(self, service: str, cost: float, is_critical: bool) -> dict:
        # Hoare-Invariant: { Cash - cost >= 0 and Runway >= 14 } Dispatch { Insolvent = False }
        metrics = self.calculate_runway()
        
        # Intercept Runaway Burn Velocity Spikes (> 3.5x trailing spend)
        trailing_1m = sum(c for ts, c in self.window_spend if time.time() - ts <= 60)
        if trailing_1m > 0 and (cost / trailing_1m) > 3.5:
            return {"verdict": "BLOCKED", "reason": "Runaway velocity spike detected"}

        if metrics["tier"] == "TIER_3_HARD_HALT":
            return {"verdict": "BLOCKED", "reason": "Emergency halt: runway < 14 days"}
            
        if metrics["tier"] == "TIER_2_DOWNGRADED" and not is_critical:
            return {"verdict": "DEGRADED", "reason": "Non-critical request throttled"}

        self.cash_reserve -= cost
        self.window_spend.append((time.time(), cost))
        return {"verdict": "APPROVED", "cost": cost}
